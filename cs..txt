/* ========================================= */
/* SLIDER */
/* ========================================= */

/*
  slider شامل تمام section ها است.

  در Desktop:

  position: fixed

  باعث می‌شود Slider روی صفحه ثابت بماند.

  سپس TypeScript با transform
  آن را به سمت چپ حرکت می‌دهد.
*/
.slider {

  /* نمایش Flex */
  display: flex;

  /* شروع از بالا سمت چپ */
  top: 0;

  left: 0;

  /* انیمیشن نرم برای transform */
  transition: transform 0.05s linear;

}


/* ========================================= */
/* SLIDE */
/* ========================================= */

/*
  هر slide یک صفحه کامل است.
*/
.slide {

  /* عرض هر section برابر عرض صفحه */
  width: 100vw;

  /* ارتفاع هر section برابر ارتفاع صفحه */
  min-height: 100vh;

  /* جلوگیری از کوچک شدن در Flexbox */
  flex-shrink: 0;

  /* فاصله داخلی */
  padding: 20px;

}
